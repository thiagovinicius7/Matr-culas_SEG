    // ===================== VIVÊNCIAS =====================
    // Colar este bloco no Firebase Console (Firestore Database → Regras),
    // logo ANTES das duas últimas chaves "}" do arquivo, e clicar em Publicar.
    // A equipe (login com e-mail e senha) faz tudo; a família, pelo link
    // público (login anônimo), só consegue VER as vivências e CRIAR a própria
    // inscrição — nunca lê a lista de inscritos de ninguém.

    function isEquipe() {
      return request.auth != null
        && request.auth.token.get('firebase', {}).get('sign_in_provider', 'anonymous') != 'anonymous';
    }

    match /vivencias/{vivenciaId} {
      allow read: if request.auth != null;
      allow create, delete: if isEquipe();
      // Família (anônima) só pode aumentar o contador de inscritos, em até 6
      // crianças por vez, e só se as inscrições estiverem abertas.
      allow update: if isEquipe() || (
        request.auth != null
        && resource.data.aberta == true
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['inscritos'])
        && request.resource.data.inscritos is int
        && request.resource.data.inscritos > resource.data.get('inscritos', 0)
        && request.resource.data.inscritos <= resource.data.get('inscritos', 0) + 6
      );
    }

    match /vivenciaInscricoes/{inscricaoId} {
      allow read, update, delete: if isEquipe();
      allow create: if isEquipe() || (
        request.auth != null
        && inscricaoId.matches('^[a-zA-Z0-9_\\-]{1,128}$')
        && request.resource.data.keys().hasOnly(['id', 'vivenciaId', 'grupoId', 'criancaNome', 'criancaNascimento', 'responsavelNome', 'parentesco', 'telefone', 'email', 'escolaAtual', 'comoConheceu', 'observacoes', 'criadoEm', 'presenca', 'origem'])
        && request.resource.data.id == inscricaoId
        && request.resource.data.vivenciaId is string
        && request.resource.data.criancaNome is string && request.resource.data.criancaNome.size() > 0 && request.resource.data.criancaNome.size() <= 200
        && request.resource.data.responsavelNome is string && request.resource.data.responsavelNome.size() <= 200
        && request.resource.data.telefone is string && request.resource.data.telefone.size() <= 40
        && (!request.resource.data.keys().hasAny(['observacoes']) || request.resource.data.observacoes.size() <= 2000)
        && request.resource.data.presenca == 'pendente'
        && request.resource.data.origem == 'link'
        && get(/databases/$(database)/documents/vivencias/$(request.resource.data.vivenciaId)).data.aberta == true
      );
    }
